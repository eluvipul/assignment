def find_median(delivery_times):
    delivery_times.sort()
    n = len(delivery_times)
    middle = n // 2

    if n % 2 == 1:
        return delivery_times[middle]
    else:
        return (delivery_times[middle - 1] +
                delivery_times[middle]) / 2


times = [32, 28, 29, 45, 30, 31, 60, 30, 29]

print("Median delivery time:", find_median(times), "minutes")

from collections import Counter

genres = [
    'Music', 'Vlog', 'Music', 'Tech', 'Music', 'Vlog',
    'Tech', 'Music', 'Comedy', 'Vlog', 'Music', 'Comedy'
]

# Count how often each genre appears
genre_counts = Counter(genres)

# Find the most watched genre
mode_genre = genre_counts.most_common(1)[0][0]

print("Genre counts:", genre_counts)
print("Most watched genre:", mode_genre)

transactions = [100, 120, 105, 110, 5000, 115, 108]

# Calculate mean
mean = sum(transactions) / len(transactions)

# Calculate median
sorted_transactions = sorted(transactions)
n = len(sorted_transactions)
median = sorted_transactions[n // 2]

print("Mean transaction amount: ₹", round(mean, 2))
print("Median transaction amount: ₹", median)
