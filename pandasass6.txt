import pandas as pd

# Example DataFrame
# df = pd.read_csv("ipl_ticket_sales.csv") 
# Columns: date, tickets_sold

# Step 1: Calculate mean and standard deviation
mean_sales = df["tickets_sold"].mean()
std_sales = df["tickets_sold"].std()

# Step 2: Compute Z-score for each row
df["z_score"] = (df["tickets_sold"] - mean_sales) / std_sales

# Step 3: Filter rows with |Z-score| > 2
outliers = df[df["z_score"].abs() > 2]

# Step 4: Print the dates of outliers
print("Dates with unusual ticket sales:")
print(outliers["date"].tolist())
