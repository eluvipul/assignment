import numpy as np

def detect_outliers(order_amounts):
    # Calculate the first and third quartiles
    Q1 = np.percentile(order_amounts, 25)
    Q3 = np.percentile(order_amounts, 75)

    # Calculate the interquartile range
    IQR = Q3 - Q1

    # Calculate the lower and upper bounds
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR

    # Identify outliers
    outliers = [
        amount for amount in order_amounts
        if amount < lower_bound or amount > upper_bound
    ]

    print("Q1:", Q1)
    print("Q3:", Q3)
    print("IQR:", IQR)
    print("Lower bound:", lower_bound)
    print("Upper bound:", upper_bound)
    print("Outlier amounts:", outliers)

# Sample Swiggy order amounts (₹)
orders = [150, 200, 180, 250, 220, 190, 210, 175, 230, 1500]

detect_outliers(orders)

import matplotlib.pyplot as plt

# Weekly screen time for 8 weeks (in hours)
screen_time = [28, 32, 30, 35, 31, 29, 33, 65]

# Create the boxplot and highlight outliers
plt.figure(figsize=(8, 5))

plt.boxplot(
    screen_time,
    patch_artist=True,
    boxprops=dict(facecolor="skyblue"),
    medianprops=dict(color="black", linewidth=2),
    flierprops=dict(
        marker="D",
        markerfacecolor="red",
        markeredgecolor="red",
        markersize=8,
        label="Outlier"
    )
)

plt.title("Weekly Screen Time Over 8 Weeks")
plt.xlabel("Screen Time")
plt.ylabel("Hours per Week")
plt.xticks([1], ["8 Weeks"])
plt.grid(axis="y", linestyle="--", alpha=0.5)

plt.tight_layout()
plt.show()
