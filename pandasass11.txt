3.import pandas as pd

# Step 1: Create daily streams data
data = {
    "user_id": [1, 1, 1, 2, 2, 2, 3, 3, 3],
    "day": ["Mon", "Tue", "Wed", "Mon", "Tue", "Wed", "Mon", "Tue", "Wed"],
    "streams": [50, 60, 55, 40, 45, 42, 70, 65, 68]

4.import pandas as pd

# Step 1: Example Zomato orders DataFrame
data = {
    "restaurant": ["Pizza Hut", "Pizza Hut", "Domino's", "Domino's", "Biryani House", "Biryani House"],
    "month": ["Jan", "Feb", "Jan", "Feb", "Jan", "Feb"],
    "total_amount": [500, 600, 450, 550, 700, 800]
}
df = pd.DataFrame(data)

print("Original DataFrame:")
print(df)

# Step 2: Use pivot_table to calculate average order amount
pivoted = df.pivot_table(
    index="restaurant",      # rows = restaurants
    columns="month",         # columns = months
    values="total_amount",   # values = order amounts
    aggfunc="mean"           # aggregation = average
)

print("\nPivoted DataFrame (Average Order Amount):")
print(pivoted)

5.import pandas as pd

# Step 1: Load monthly sales CSVs
# (Replace with actual file paths if you have them)
sales_jan = pd.DataFrame({
    "product": ["Shoes", "T-Shirt", "Jeans"],
    "sales": [5000, 3000, 4000]
})

sales_feb = pd.DataFrame({
    "product": ["Shoes", "T-Shirt", "Jeans"],
    "sales": [6000, 3500, 4200]
})

sales_mar = pd.DataFrame({
    "product": ["Shoes", "T-Shirt", "Jeans"],
    "sales": [5500, 3200, 4100]
})

# Step 2: Concatenate row-wise
all_sales = pd.concat([sales_jan, sales_feb, sales_mar], ignore_index=True)

print("Combined DataFrame:")
print(all_sales)

# Step 3: Group by product and sum sales
total_sales= all_sales.groupby("product")["sales"].sum().reset_index()

print("\nTotal Sales per Product:")
print(total_sales)
