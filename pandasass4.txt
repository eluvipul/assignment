You have a DataFrame of Zomato restaurant names and their ratings. Use reindex() to rearrange the rows so that your favorite restaurant appears first, followed by the rest in any order.

1.import pandas as pd

# Step 1: Create the DataFrame
movies = pd.DataFrame({
    "title": [
        "3 Idiots",
        "Dangal",
        "Sholay",
        "Gully Boy",
        "Kabir Singh",
        "Zindagi Na Milegi Dobara"
    ],
    "rating": [8.4, 8.3, 8.2, 7.9, 7.3, 8.1],
    "release_year": [2009, 2016, 1975, 2019, 2019, 2011]
})

# Step 2: Sort by rating (descending)
sorted_movies = movies.sort_values(by="rating", ascending=False)

# Step 3: Display the result
print(sorted_movies)

5.import pandas as pd

# Example DataFrame
df = pd.DataFrame({
    "product_name": ["Laptop", "Smartphone", "Shoes", "Watch", "Headphones"],
    "price": [60000, 30000, 2000, 5000, 1500],
    "discount": [20, 15, 40, 10, 25]
})

# Step 1: Sort by discount descending
df_sorted = df.sort_values(by="discount", ascending=False)

# Step 2: Reset index to start from 0
df_reset = df_sorted.reset_index(drop=True)

# Step 3: Print the result
print(df_reset)
