2.

import pandas as pd

# Step 1: Example Zomato orders DataFrame
data = {
    "city": ["Mumbai", "Mumbai", "Delhi", "Delhi", "Bangalore", "Bangalore"],
    "restaurant": ["Spice Villa", "Spice Villa", "Curry House", "Curry House", "Tandoori Hub", "Tandoori Hub"],
    "order_amount": [500, 700, 300, 400, 600, 800]
}
df = pd.DataFrame(data)

print("Original DataFrame:")
print(df)

# Step 2: Group by city and restaurant, then calculate average order_amount
avg_orders = df.groupby(["city", "restaurant"]).agg(
    avg_order_amount=("order_amount", "mean")
).reset_index()

print("\nAverage Order Amount per Restaurant in Each City:")
print(avg_orders)

3.import pandas as pd

# Step 1: Example Flipkart-style sales DataFrame
data = {
    "category": ["Electronics", "Electronics", "Fashion", "Fashion", "Books", "Books"],
    "region": ["North", "South", "North", "South", "North", "South"],
    "sales": [60000, 45000, 20000, 25000, 12000, 15000],
    "units_sold": [300, 250, 400, 350, 500, 600]
}
df = pd.DataFrame(data)

print("Original DataFrame:")
print(df)

# Step 2: Group by category and region, apply multiple aggregations
agg_result = df.groupby(["category", "region"]).agg(
    total_sales=("sales", "sum"),
    avg_units_sold=("units_sold", "mean")
).reset_index()

print("\nAggregated Result:")
print(agg_result)

4.import pandas as pd

# Step 1: Example DataFrame
data = {
    "region": ["North", "North", "South", "South", "East", "East", "West", "West"],
    "revenue": [5000, 7000, 8000, 6000, 4000, 4500, 9000, 8500]
}
df = pd.DataFrame(data)

print("Original DataFrame:")
print(df)

# Step 2: Group by region and apply multiple aggregations
region_stats = df.groupby("region").agg(
    total_revenue=("revenue", "sum"),
    max_revenue=("revenue", "max"),
    min_revenue=("revenue", "min")
)

print("\nAggregated Result:")
print(region_stats)
