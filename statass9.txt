import numpy as np
import matplotlib.pyplot as plt

# Parameters of the normal distribution
mean = 500
std_dev = 100
size = 1000

# Generate 1000 random numbers
data = np.random.normal(loc=mean, scale=std_dev, size=size)

# Plot the distribution as a histogram
plt.figure(figsize=(9, 5))
plt.hist(data, bins=30, color="skyblue", edgecolor="black",
         alpha=0.75)

plt.axvline(mean, color="red", linestyle="--",
            linewidth=2, label="Mean = 500")

plt.title("Normal Distribution of 1000 Random Numbers")
plt.xlabel("Value")
plt.ylabel("Frequency")
plt.legend()
plt.grid(axis="y", linestyle=":", alpha=0.5)

plt.tight_layout()
plt.show()

# Display sample statistics
print("Sample mean:", round(np.mean(data), 2))
print("Sample standard deviation:", round(np.std(data), 2))

import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# Standard normal distribution: mean = 0, standard deviation = 1
mu = 0
sigma = 1

# Generate x values and corresponding probability density
x = np.linspace(-4, 4, 1000)
y = norm.pdf(x, mu, sigma)

# Plot the normal curve
plt.figure(figsize=(9, 5))
plt.plot(x, y, color="blue", linewidth=2, label="Standard normal curve")

# Shade the area between Z = -1 and Z = 1
x_shade = np.linspace(-1, 1, 300)
y_shade = norm.pdf(x_shade, mu, sigma)
plt.fill_between(x_shade, y_shade, color="skyblue",
                 alpha=0.7, label="Area between Z = -1 and Z = 1")

# Mark the Z boundaries
plt.axvline(-1, color="red", linestyle="--", label="Z = -1")
plt.axvline(1, color="red", linestyle="--", label="Z = 1")

plt.title("Standard Normal Distribution")
plt.xlabel("Z-score")
plt.ylabel("Probability Density")
plt.legend()
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

# Calculate the probability
probability = norm.cdf(1) - norm.cdf(-1)
print(f"Probability between Z = -1 and Z = 1: {probability:.4f}")
print(f"Percentage: {probability * 100:.2f}%")

# Given parameters
mean = 30
std_dev = 5

print(f"Percentage: {probability * 100:.2f}%")

## 1\. Prompt given to ChatGPT

Writing

Explain in simple words why the normal distribution is important for analyzing real-world app data, such as sales, food delivery times, or user ratings. Give practical examples and explain how businesses can use it to make better decisions.

## 2\. AI's answer

The normal distribution is important because it helps businesses understand how data is spread around an average value. It usually looks like a bell-shaped curve, where most values are near the average and fewer values are extremely high or low.

For example:

- Sales: An e-commerce app like Flipkart can analyze daily sales to understand typical sales levels and identify unusually high or low sales days.
- Delivery times: A food delivery app like Swiggy can study delivery times to estimate how long orders usually take and identify unusually delayed deliveries.
- User ratings: An app can analyze restaurant ratings to understand overall customer satisfaction and identify ratings that differ significantly from the average.

The normal distribution also helps businesses calculate probabilities, identify unusual observations using Z-scores, and make predictions based on historical data.

However, not all real-world data follows a normal distribution. For example, sales amounts may be strongly skewed, and ratings are often limited to values such as 1 to 5 stars. Businesses should check the shape of their data before assuming it is normally distributed.

## 3\. Conclusion

The normal distribution helps businesses understand typical patterns, detect unusual events, and make better data-driven decisions about sales, delivery performance, and customer satisfaction.
