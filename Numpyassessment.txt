Assessment 2: Automobile Sales Dataset

1.Line chart of sales over years

import pandas as pd
import matplotlib.pyplot as plt

# Load dataset
df = pd.read_csv("historical_automobile_sales.csv")

# Group by Year and sum sales
sales_by_year = df.groupby('Year')['Automobile_Sales'].sum()

# Plot line chart
sales_by_year.plot(kind='line', figsize=(10,6), marker='o')

plt.title("Automobile Sales Over Years")
plt.xlabel("Year")
plt.ylabel("Total Sales")
plt.grid(True)
plt.show()

2.Different lines for vehicle types

df.groupby(['Year','Vehicle_Type'])['Automobile_Sales'].sum().unstack().plot(figsize=(12,6))
plt.title("Sales Trends by Vehicle Type")
plt.ylabel("Sales")
plt.show()

3.Seaborn comparison recession vs non-recession

import seaborn as sns

sns.lineplot(data=df, x='Year', y='Automobile_Sales', hue='Vehicle_Type', style='Recession')
plt.title("Sales Trend per Vehicle Type (Recession vs Non-Recession)")
plt.show()

4.Bar chart comparing recession vs non-recession

sns.barplot(
    data=df,
    x='Vehicle_Type',
    y='Automobile_Sales',
    hue='Recession',
    estimator='mean'
)
plt.title("Average Sales by Vehicle Type (Recession vs Non-Recession)")
plt.show()
