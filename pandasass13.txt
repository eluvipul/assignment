2.import pandas as pd

# Step 1: Load the dataset
df = pd.read_csv("IPL_2023_Batsmen.csv")

# Step 2: Perform univariate analysis on 'runs'
mean_runs = df["runs"].mean()
median_runs = df["runs"].median()
mode_runs = df["runs"].mode()[0]   # mode() returns a Series, take the first value
min_runs = df["runs"].min()
max_runs = df["runs"].max()

# Step 3: Print results
print("Univariate Analysis of 'runs' column:")
print(f"Mean runs   : {mean_runs}")
print(f"Median runs : {median_runs}")
print(f"Mode runs   : {mode_runs}")
print(f"Minimum runs: {min_runs}")
print(f"Maximum runs: {max_runs}")
