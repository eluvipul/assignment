import pandas as pd
import numpy as np

# Create a sample dataset of 1000 Flipkart orders
np.random.seed(42)

cities = ["Ahmedabad", "Surat", "Vadodara"]
city_probabilities = [0.5, 0.3, 0.2]

orders = pd.DataFrame({
    "Order_ID": range(1, 1001),
    "City": np.random.choice(
        cities, size=1000, p=city_probabilities
    )
})

# Calculate proportional sample sizes for each city
sample_size = 60
city_counts = orders["City"].value_counts()
city_proportions = city_counts / len(orders)
sample_counts = (city_proportions * sample_size).round().astype(int)

# Adjust rounding so the total sample size is exactly 60
while sample_counts.sum() < sample_size:
    fractions = city_proportions * sample_size - sample_counts
    sample_counts.loc[fractions.idxmax()] += 1

while sample_counts.sum() > sample_size:
    fractions = city_proportions * sample_size - sample_counts
    sample_counts.loc[fractions.idxmin()] -= 1

# Select orders proportionally from each city
stratified_sample = pd.concat([
    orders[orders["City"] == city].sample(
        n=int(sample_counts[city]), random_state=42
    )
    for city in sample_counts.index
], ignore_index=True)

# Display results
print("Sample size:", len(stratified_sample))
print("\nOrders sampled per city:")
print(stratified_sample["City"].value_counts())
print("\nSelected orders:")
print(stratified_sample.head())

Python pseudocode / function

python

def detect_scale(data, column_name=""):
    # Remove missing values
    values = [v for v in data if v is not None]

    if not values:
        return "Unknown"

    name = column_name.lower()

    # Rule 1: Known ordered categories
    ordinal_keywords = ["order status", "size", "rating",
                        "priority", "delivery stage"]

    if any(word in name for word in ordinal_keywords):
        return "Ordinal"

    # Rule 2: Dates and times
    if all(isinstance(v, str) for v in values):
        if "date" in name or "time" in name:
            return "Interval"

        # Categories with a known order
        ordered = {
            "order status": ["placed", "packed", "shipped",
                             "out for delivery", "delivered"],
            "rating": ["1 star", "2 stars", "3 stars",
                       "4 stars", "5 stars"]
        }

        for key, order in ordered.items():
            if key in name:
                return "Ordinal"

        return "Nominal"

    # Rule 3: Numerical columns
    if all(isinstance(v, (int, float)) and not isinstance(v, bool)
           for v in values):

        # Counts and amounts generally have a true zero
        ratio_keywords = ["price", "quantity", "count",
                          "amount", "discount", "weight"]

        if any(word in name for word in ratio_keywords):
            return "Ratio"

        # Temperature or calendar-year-like values
        if "temperature" in name or "year" in name:
            return "Interval"

        # Default for unexplained numerical values
        return "Ratio"

    # Rule 4: Other values
    return "Nominal"

def find_median(delivery_times):
    delivery_times.sort()
    n = len(delivery_times)
    middle = n // 2

    if n % 2 == 1:
        return delivery_times[middle]
    else:
        return (delivery_times[middle - 1] +
                delivery_times[middle]) / 2

city_probabilities = [0.5, 0.3, 0.2]

orders = pd.DataFrame({
    "Order_ID": range(1, 1001),
    "City": np.random.choice(
        cities, size=1000, p=city_probabilities
    )
})

# Calculate proportional sample sizes for each city
sample_size = 60
city_counts = orders["City"].value_counts()
city_proportions = city_counts / len(orders)
sample_counts = (city_proportions * sample_size).round().astype(int)

# Adjust rounding so the total sample size is exactly 60
while sample_counts.sum() < sample_size:
    fractions = city_proportions * sample_size - sample_counts
    sample_counts.loc[fractions.idxmax()] += 1

while sample_counts.sum() > sample_size:

random.seed(42)

# Create a sample dataset of 200 IPL fans
teams = ["CSK", "MI", "RCB", "KKR", "RR"]
fans = pd.DataFrame({
    "Fan_ID": range(1, 201),
    "Favorite_Team": (
        ["CSK"] * 60 +
        ["MI"] * 50 +
        ["RCB"] * 40 +
        ["KKR"] * 30 +
        ["RR"] * 20
    )
})

# Group fans by their favorite team
team_groups = defaultdict(list)
for _, fan in fans.iterrows():
    team_groups[fan["Favorite_Team"]].append(fan.to_dict())

# Calculate proportional sample sizes
total_fans = len(fans)
sample_size = 40
quotas = {
    team: len(group) * sample_size / total_fans
    for team, group in team_groups.items()
}

# Round quotas using the largest-remainder method
sample_counts = {
    team: int(quota) for team, quota in quotas.items()
}
remaining = sample_size - sum(sample_counts.values())

teams_by_remainder = sorted(
    quotas,
    key=lambda team: quotas[team] - sample_counts[team],
    reverse=True
)

for team in teams_by_remainder[:remaining]:
    sample_counts[team] += 1

# Randomly select fans within each team
selected_fans = []
