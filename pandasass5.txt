# Example DataFrame

3.import pandas as pd

# Example DataFrame
df = pd.DataFrame({
    "product_name": ["Laptop", "Smartphone", "Shoes", "Watch", "Headphones"],
    "rating": [4.5, None, 3.8, None, 4.2],
    "review": ["Great performance", "Good value", "Comfortable", "Stylish", "Clear sound"]
})

# Step 1: Calculate mean rating (ignores NaN by default)
mean_rating = df["rating"].mean()

# Step 2: Fill missing ratings with mean
df["rating"] = df["rating"].fillna(mean_rating)

# Step 3: Display updated DataFrame
print(df)

4.import pandas as pd

5.import pandas as pd

# Step 1: Create sample Instagram user data
df = pd.DataFrame({
    "username": ["user1", "user2", "user3", "user4", "user5"],
    "followers": [1200, None, 560, None, 300],
    "posts": [50, 20, 35, 10, 5]
})
