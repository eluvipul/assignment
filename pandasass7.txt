3.import pandas as pd

# Sample DataFrame
data = {
    "PostID": [101, 102, 103, 104],
    "Likes": [1500, 2500000, 950, 12000],
    "Comments": [120, 4500, 75, 300000]
}
df = pd.DataFrame(data)

# Step 1: Define helper function
def format_instagram_number(x):
    if x >= 1_000_000:
        return f"{x/1_000_000:.1f}M"
    elif x >= 1_000:
        return f"{x/1_000:.1f}K"
    else:
        return str(x)

# Step 2: Apply formatting with applymap on selected columns
df[["Likes", "Comments"]] = df[["Likes", "Comments"]].applymap(format_instagram_number)

# Step 3: Display updated DataFrame
print(df)


4.import pandas as pd

# Example DataFrame
data = {
    "Restaurant": ["Pizza Hut", "Domino's", "Biryani House", "BBQ Nation"],
    "Price": [500, 300, 250, 800],
    "Qty": [2, 3, 4, 1],
    "DeliveryCharge": [50, 40, 60, 100]
}
df = pd.DataFrame(data)

# Step 1: Use apply with lambda to calculate FinalAmount
df["FinalAmount"] = df.apply(lambda row: (row["Price"] * row["Qty"]) + row["DeliveryCharge"], axis=1)

# Step 2: Display updated DataFrame
print(df)
