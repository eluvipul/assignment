import pandas as pd

# Step 1: Create restaurants DataFrame
restaurants = pd.DataFrame({
    "restaurant_id": [1, 2, 3],
    "name": ["Spice Villa", "Curry House", "Tandoori Hub"],
    "city": ["Mumbai", "Delhi", "Bangalore"]
})

# Step 2: Create zomato_reviews DataFrame
zomato_reviews = pd.DataFrame({
    "review_id": [101, 102, 103, 104],
    "restaurant_id": [1, 2, 4, 5],  # 4 and 5 don't exist in restaurants
    "rating": [4.5, 4.0, 3.5, 5.0],
    "reviewer_city": ["Mumbai", "Delhi", "Hyderabad", "Chennai"]
})

# Step 3: Perform outer join
merged_df = pd.merge(restaurants, zomato_reviews, on="restaurant_id", how="outer")

# Step 4: Display merged DataFrame
print(merged_df)

4.import pandas as pd

# Step 1: Create playlists DataFrame
playlists = pd.DataFrame({
    "playlist_id": [301, 302, 303, 304],
    "user_id": [1, 2, 3, 4],
    "genre": ["Rock", "Pop", "Jazz", "Classical"],
    "city": ["Delhi", "Mumbai", "Bangalore", "Chennai"]  # playlist owner's city
})

# Step 2: Create user_profiles DataFrame
user_profiles = pd.DataFrame({
    "user_id": [1, 2, 3, 5],
    "username": ["Alice", "Bob", "Charlie", "Eva"],
    "city": ["Delhi", "Mumbai", "Bangalore", "Kolkata"]  # user profile city

# Step 3: Merge on both user_id and city
merged_df = pd.merge(playlists, user_profiles, on=["user_id", "city"], how="inner")

# Step 4: Display merged DataFrame
print(merged_df)


5.import pandas as pd

# Step 1: Create Flipkart products DataFrame
products = pd.DataFrame({
    "product_id": [1, 2, 3],
    "name": ["Smartphone", "Laptop", "Headphones"],
    "category": ["Electronics", "Electronics", "Accessories"]
})

# Step 2: Create product ratings DataFrame
ratings = pd.DataFrame({
    "rating_id": [101, 102, 103, 104],
    "product_id": [1, 2, 4, 5],  # Notice 4 and 5 don't exist in products
    "rating": [4.5, 4.0, 3.5, 5.0]
})

# Step 3: Default merge (inner join)
default_merge = pd.merge(products, ratings, on="product_id")
print("Default (Inner Join):")
print(default_merge)

# Step 4: Modified merge (right join)
right_merge = pd.merge(products, ratings, on="product_id", how="right")
print("\nRight Join:")
print(right_merge)
