4.import pandas as pd

# Step 1: Example IPL matches DataFrame (already loaded as ipl_df)
# Columns include: ['date', 'season', 'team1', 'team2', 'winner', 'win_by_runs', 'win_by_wickets']

# Step 2: Create a DataFrame of team names and their home cities
teams_data = {
    "team": ["Mumbai Indians", "Chennai Super Kings", "Royal Challengers Bangalore", 
             "Kolkata Knight Riders", "Rajasthan Royals"],
    "home_city": ["Mumbai", "Chennai", "Bangalore", "Kolkata", "Jaipur"]
}
teams_df = pd.DataFrame(teams_data)

# Step 3: Merge to add home city for team1
ipl_df = ipl_df.merge(teams_df, left_on="team1", right_on="team", how="left")
ipl_df = ipl_df.rename(columns={"home_city": "team1_home_city"})
ipl_df = ipl_df.drop(columns=["team"])

# Step 4: Merge to add home city for team2
ipl_df = ipl_df.merge(teams_df, left_on="team2", right_on="team", how="left")
ipl_df = ipl_df.rename(columns={"home_city": "team2_home_city"})
ipl_df = ipl_df.drop(columns=["team"])

# Step 5: Display first 5 rows
print(ipl_df.head())


5.import pandas as pd

# Step 1: Create pivot table of matches won per team per season
pivot_table = pd.pivot_table(
    ipl_df,
    index="season",
    columns="winner",
    values="date",   # any column works, we just need counts
    aggfunc="count",
    fill_value=0
)

print("Pivot Table: Matches Won by Each Team per Season")
print(pivot_table.head())

# Step 2: Find the top 3 teams with most wins in any single season
# Stack to long format, then sort
wins_per_season = pivot_table.stack().reset_index()
wins_per_season.columns = ["season", "team", "wins"]

top3 = wins_per_season.sort_values(by="wins", ascending=False).head(3)

print("\nTop 3 Teams with Most Wins in a Single Season Season:")
print(top3)
