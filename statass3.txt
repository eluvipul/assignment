2.

import statistics
import math

# Order amounts in rupees
orders = [250, 400, 350, 300, 500]

# Calculate mean
mean = sum(orders) / len(orders)

# Calculate population variance
variance = sum((x - mean) ** 2 for x in orders) / len(orders)

# Calculate standard deviation
std_dev = math.sqrt(variance)

print("Mean:", mean)
print("Variance:", variance)
print("Standard deviation:", round(std_dev, 2))

3.

import statistics

# Monthly mobile data usage in GB
data_usage = [3.2, 2.8, 3.5, 3.0, 2.9, 3.1]

# Calculate mean and population standard deviation
mean = statistics.mean(data_usage)
std_dev = statistics.pstdev(data_usage)

# Calculate coefficient of variation
cv = (std_dev / mean) * 100

print("Mean:", round(mean, 2), "GB")
print("Standard Deviation:", round(std_dev, 2), "GB")
print("Coefficient of Variation:", round(cv, 2), "%")

import statistics

# Likes on 10 recent Instagram posts
likes = [120, 150, 130, 110, 170, 140, 135, 160, 125, 155]

# Calculate mean and population standard deviation
mean = statistics.mean(likes)
std_dev = statistics.pstdev(likes)

# Compare standard deviation with mean
cv = (std_dev / mean) * 100

print("Mean likes:", round(mean, 2))
print("Standard deviation:", round(std_dev, 2))
print("Coefficient of variation:", round(cv, 2), "%")
